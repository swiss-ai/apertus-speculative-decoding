#!/usr/bin/env python3
"""Package the two completed T1 sweeps without changing their raw artifacts.

Run on Clariden with Python 3.12. Output is a new recovery archive; weights and
source corpora are never included. Recovery-time source evidence is labelled
separately from the immutable as-run files.
"""
from __future__ import annotations

import datetime as dt
import hashlib
import io
import json
from pathlib import Path
import subprocess
import tarfile
import tomllib

REPO = Path("/iopsstor/scratch/cscs/faruk_zahiragic/spec-exp/waste-profiling")
RUNS = ["waste-sum-a-20261024T133232Z", "waste-sum-b-20261024T133235Z"]
OUTPUT = Path("/iopsstor/scratch/cscs/faruk_zahiragic/spec-exp-data/waste-profiling/recovery-20261027")
PREFIX = Path("results/8b/waste/recovery-20261027/source-recovery")


def main() -> None:
    OUTPUT.mkdir(parents=True, exist_ok=True)
    archive = OUTPUT / "completed-summarization-source-v1.tar.gz"
    if archive.exists():
        raise FileExistsError(archive)
    manifest = {
        "collected_at": dt.datetime.now(dt.timezone.utc).isoformat(),
        "source_repo": str(REPO),
        "recovery_git_revision": subprocess.check_output(["git", "-C", str(REPO), "rev-parse", "HEAD"], text=True).strip(),
        "claim_limit": "Original metadata is unchanged. Recovery git/source identities are not contemporaneous launch receipts.",
        "source_files": [],
        "recovery_files": [],
        "unavailable_recovery_sources": [],
    }
    seen = set()
    with tarfile.open(archive, "w:gz") as tar:
        def add(path: Path, destination: Path, recovery: bool = False) -> None:
            key = str(destination)
            if key in seen:
                return
            seen.add(key)
            if not path.is_file():
                manifest["unavailable_recovery_sources"].append(str(path))
                return
            data = path.read_bytes()
            info = tarfile.TarInfo(key)
            info.size = len(data)
            info.mtime = int(path.stat().st_mtime)
            tar.addfile(info, io.BytesIO(data))
            manifest["recovery_files" if recovery else "source_files"].append({
                "source": str(path), "path": key, "bytes": len(data),
                "sha256": hashlib.sha256(data).hexdigest(),
            })

        for name in RUNS:
            run = REPO / "results/8b/sweeps" / name
            for path in sorted(run.rglob("*")):
                if path.is_file():
                    add(path, path.relative_to(REPO))
            for line in (run / "patches.txt").read_text().splitlines():
                path = Path(line.strip())
                add(path, PREFIX / "source-patches" / path.name, True)
            environment = tomllib.loads((run / "environment.toml").read_text())
            image = Path(environment["image"])
            manifest.setdefault("container_recovery_stat", {})[str(image)] = {
                "bytes": image.stat().st_size, "mtime_ns": image.stat().st_mtime_ns,
                "claim_limit": "No contemporaneous image digest was found in these sweep records.",
            }
            for mount in environment.get("mounts", []):
                source = Path(mount.split(":", 1)[0])
                if "vllm-patch-sweep-" in str(source) and source.is_file():
                    relative = source.parts[source.parts.index("workspace"):]
                    add(source, PREFIX / "resolved-overlays" / name / Path(*relative), True)
        for relative in [
            "targets/8b/contract.json", "methods/dspark/scheduler/sts-dspark-e10-8b.json",
            "serving/node-sweep.sh", "src/apertus_bench/node_sweep.py", "src/apertus_bench/waste_report.py",
        ]:
            add(REPO / relative, PREFIX / "recovery-checkout" / relative, True)
        for relative in [
            "logs/image.sha256", "provenance-3737473.json",
            "candidates/opb-0.summary.json",
        ]:
            source = Path("/iopsstor/scratch/cscs/faruk_zahiragic/spec-exp-data/tree-drafters/runs/full-20261024T1000Z") / relative
            add(source, PREFIX / "other-run-identity-reference" / relative, True)
        data = (json.dumps(manifest, indent=2) + "\n").encode()
        info = tarfile.TarInfo(str(PREFIX / "collection-manifest.json"))
        info.size = len(data)
        tar.addfile(info, io.BytesIO(data))
    digest = hashlib.sha256(archive.read_bytes()).hexdigest()
    (OUTPUT / "archive.sha256").write_text(f"{digest}  {archive.name}\n")
    print(json.dumps({"archive": str(archive), "sha256": digest, "bytes": archive.stat().st_size,
                      "source_files": len(manifest["source_files"]), "recovery_files": len(manifest["recovery_files"])}))


if __name__ == "__main__":
    main()
